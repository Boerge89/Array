public void kuehlsteWaermsteStunde() {
  int minTemperatur = tag[0]; int maxTemperatur = tag[0];
  int minStunde = 0; int maxStunde = 0;
  for (int i = 1; i < tag.length; i++) {
    if (tag[i] < minTemperatur) {
      minTemperatur = tag[i]; minStunde = i;
    }
    if (tag[i] > maxTemperatur) {
      maxTemperatur = tag[i]; maxStunde = i;
    }
  }
}
